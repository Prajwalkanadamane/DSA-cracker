Heap is a complete binary tree that comes with a heap order property!

Binary tree: at most two child nodes.

CBT: every level is completely filled except possibly  the last level.
    ->Nodes always added from left(lean towards the left).

Heap order Property:
1.Max heap: child nodes values are always lesser than it's parents.
2.Min heap: vice-versa.

class heap{
    public:

    int arr[100];
    int size;

    heap(){
        arr[0] = -1;
        size = 0;
    }

    void insert(int val){
        size = size+1;
        int index = size;
        arr[index] = val;

        while(index > 1){
            int parent = index/2;

            if (arr[parent]  < arr[index]){
                swap(arr[parent], arr[index]);
                index = parent;
            }else{
                return;
            }
        }
    }

    void print(){
        for (int i=1; i<=size; i++){
            cout << arr[i] << " ";
        }
    }

    void delete(){
        if (size == 0) cout << "Mothing to deltete"; return;

        arr[1] = arr[size]; // put last element into first index
        size--; // remove last element

        //take root node to its correct position
        int i=1;
        while(i < size){
            int leftIdx = 2*i;
            int rightIdx = 2*i+1;

            if (leftIdx < size && arr[i] < arr[leftIdx] && arr[leftIdx] > arr[rightIdx]){ // we are checking out of bound condition so that we dont end up checking the left or right nodes for the leaf node.
                swap(arr[i], arr[leftIdx]);
                i = leftIdx;
            }else if (rightIdx < size && arr[i] < arr[rightIdx] && arr[leftIdx] < arr[rightIdx]){ // extra "&& arr[leftIdx] > arr[rightIdx]" this condition just to make sure we are swapping the root with the bigger child.
                swap(arr[i], arr[rightIdx]);
                i = rightIdx;
            }else{
                return;
            }
        }
    }

};

    h.insert(55);
    h.insert(53);
    h.insert(52);
    h.insert(54);
    h.print();
}

Heapify Algorithm:
In complete binary tree: leaf nodes are from (n/2+1) -> n.
As leaf nodes are heaps by themself we don't have to process them while performing heapify algorithm. So, we just need to process the heapify form 1 -> n/2 (process: taking the nodes to correct positions so that heap order property satisfies).

-> taking a node and placing it in the correct is what heapify(that we are gonna write) functions job is!

void heapify(int arr[], int n, int i){
    int largest = i;
    int left = 2*i, right = 2*i+1;

    if (left < n && arr[largest] < arr[left]){
        largest = left;
    }
    if (right < n && arr[largest] < arr[right]){
        largest = right;
    }

    if (largest != i){
        swap(arr[largest], arr[i]);
        heapify(arr, n, largest);
    }
}

int main(){
    int arr[6] = {-1, 54, 53, 52, 56, 50};

    int n = 5; // size of the array;
    for (int i=n/2; i>0; i--){
        heapify(arr, n, i);
    }
    cout << "Printing the array now: " << endl;
    for(int i=1; i<=n; i++){
        cout << arr[i] << " ";
    }cout << endl;
    return 0;
}

Heap Sort:
step 1: swap the first node(root) and last nodes a[1] and a[n]
step 2: take the new root node to the correct position and size--.
step 3: repeat the 1st step.
step 4: repeat the 2nd step.

void heapSort(int arr[], int n){
    int size = n;
    while(size > 1){
        swap(arr[size], arr[1]);
        size--;

        heapify(arr, size, 1);
    }
}
